SET(SHARED_TARGET OpenMMPGMCUDA)
SET(COMMON_DIR ${CMAKE_SOURCE_DIR}/platforms/common)
FILE(GLOB src_files ${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp ${COMMON_DIR}/src/*.cpp)
SET(COMMON_KERNELS_CPP ${CMAKE_BINARY_DIR}/platforms/common/src/CommonPGMKernelSources.cpp)
SET_SOURCE_FILES_PROPERTIES(${COMMON_KERNELS_CPP} PROPERTIES GENERATED TRUE)
INCLUDE_DIRECTORIES(BEFORE ${CMAKE_CURRENT_SOURCE_DIR}/src ${COMMON_DIR}/src ${CMAKE_BINARY_DIR}/platforms/common/src ${PGM_CORE_DIR})
# the engine (core/cuda/pgm_cuda.cu), static, linked into the plugin
ADD_LIBRARY(PGMCudaEngine STATIC ${PGM_CORE_DIR}/cuda/pgm_cuda.cu)
SET_TARGET_PROPERTIES(PGMCudaEngine PROPERTIES POSITION_INDEPENDENT_CODE ON CUDA_STANDARD 17)
TARGET_INCLUDE_DIRECTORIES(PGMCudaEngine PRIVATE ${PGM_CORE_DIR} ${PGM_CORE_DIR}/cuda)
TARGET_LINK_LIBRARIES(PGMCudaEngine PRIVATE CUDA::cufft CUDA::cudart)
INCLUDE_DIRECTORIES(${PGM_CORE_DIR}/cuda)
ADD_LIBRARY(${SHARED_TARGET} SHARED ${src_files} ${COMMON_KERNELS_CPP})
ADD_DEPENDENCIES(${SHARED_TARGET} PGMCommonKernels)
TARGET_INCLUDE_DIRECTORIES(${SHARED_TARGET} PRIVATE ${CUDAToolkit_INCLUDE_DIRS})
TARGET_LINK_LIBRARIES(${SHARED_TARGET} OpenMM OpenMMCUDA ${SHARED_PGM_TARGET} PGMCudaEngine CUDA::cuda_driver CUDA::cudart CUDA::cufft)
SET_TARGET_PROPERTIES(${SHARED_TARGET} PROPERTIES COMPILE_FLAGS "-DOPENMM_BUILDING_SHARED_LIBRARY")
# the toolkit's cuFFT / runtime at run time (the conda env has other versions)
SET_TARGET_PROPERTIES(${SHARED_TARGET} PROPERTIES INSTALL_RPATH "${CUDAToolkit_LIBRARY_DIR}")
INSTALL(TARGETS ${SHARED_TARGET} DESTINATION lib/plugins)
