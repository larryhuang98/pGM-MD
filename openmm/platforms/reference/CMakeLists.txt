SET(SHARED_TARGET OpenMMPGMReference)
FILE(GLOB src_files ${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp)
FILE(GLOB incl_files ${CMAKE_CURRENT_SOURCE_DIR}/src/*.h)
INCLUDE_DIRECTORIES(BEFORE ${CMAKE_CURRENT_SOURCE_DIR}/src ${PGM_CORE_DIR})
ADD_LIBRARY(${SHARED_TARGET} SHARED ${src_files} ${incl_files} ${PGM_CORE_DIR}/pgm_core.cpp)
TARGET_LINK_LIBRARIES(${SHARED_TARGET} OpenMM ${SHARED_PGM_TARGET})
SET_TARGET_PROPERTIES(${SHARED_TARGET} PROPERTIES COMPILE_FLAGS "-DOPENMM_BUILDING_SHARED_LIBRARY")
INSTALL(TARGETS ${SHARED_TARGET} DESTINATION lib/plugins)
